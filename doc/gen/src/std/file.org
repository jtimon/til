* File
:PROPERTIES:
:CUSTOM_ID: til.src.std.file.File
:group: Standard library
:END:

File handle for reading or writing on disk.

#+begin_src til
struct File {
    ref mut handle : U8
    mut is_write : Bool
    mut is_update : Bool
namespace:
    /// Read the entire file at `path` into a Str. Exits with an error if
    /// the file cannot be opened.
    proc readfile(path: Str) returns Str
    /// Write `content` to the file at `path`, replacing any previous
    /// contents. Exits with an error if the file cannot be opened.
    proc writefile(path: Str, content: Str)
    /// Append `content` to the file at `path`, keeping what is already
    /// there. The file is created when it does not exist.
    proc appendfile(path: Str, content: Str)
    /// Delete the file at `path`, like `rm`. Removing a file the process
    /// still has open is allowed on POSIX (the handle keeps working until
    /// it is closed); a missing path is not an error. Directories are
    /// not removed -- pointing this at one is an error.
    proc remove(path: Str)
    /// True when `path` names an existing directory entry. A directory,
    /// or a file the process has no permission to read, still exists.
    proc exists(path: Str) returns shallow Bool
    /// True when `path` names a directory. False for a regular file, and
    /// false for a path that does not exist -- "is a directory" is one
    /// question, "exists" is another, and `File.exists` answers that one.
    proc is_dir(path: Str) returns shallow Bool
    /// True when `path` names a regular file -- not a directory, and not
    /// a device, fifo or socket either.
    proc is_file(path: Str) returns shallow Bool
    /// Remove `path` and everything under it, like `rm -rf`: a file, or a
    /// directory with its whole subtree. A path that does not exist is
    /// already removed and reports success; a symlink is unlinked, never
    /// followed. Any other failure is fatal.
    proc remove_tree(path: Str)
    /// Move `src` onto `dst`, replacing whatever `dst` was. On POSIX the
    /// replacement is atomic, which is what makes it safe to build a new
    /// executable beside the running one and swap it in. Fatal on
    /// failure -- notably when the two paths are on different file
    /// systems, where a rename cannot happen and a copy is the answer.
    proc rename(src: Str, dst: Str)
    /// Open a file at `path` for reading or writing.
    proc new(path: Str, shallow is_write: Bool) returns File
    /// Open a file at `path` for in-place update: reads and writes are
    /// both allowed and the cursor can be moved with `seek`, so small
    /// regions can be patched without rewriting the whole file. The file
    /// must already exist.
    proc update(path: Str) returns File
    /// Open a file at `path` for appending: writes always land at the end
    /// and existing contents are kept, where `File.new(path, true)`
    /// truncates. The file is created when it does not exist. Reads are
    /// rejected, and `seek` does not move where a write lands -- "ab"
    /// appends regardless of the cursor.
    proc append(path: Str) returns File
    /// Write a string to the file and return `self` so writes chain via
    /// UFCS: `f.write("a").write("b").write("c")`. Panics unless opened
    /// for writing or update.
    proc write(self: File, s: Str) returns ref File
    /// Read the entire file contents into a Str. Panics if opened
    /// write-only.
    proc read_all(self: File) returns Str
    /// Read up to `count` bytes starting at the current cursor position
    /// and advance the cursor. The result is shorter than `count` near
    /// end-of-file. Panics if opened write-only.
    proc read(self: File, shallow count: I64) returns Str
    /// Return the current cursor position as a byte offset from the start
    /// of the file.
    proc tell(self: File) returns shallow I64
    /// Move the cursor to an absolute byte offset from the start of the
    /// file. Returns `self` so calls chain via UFCS.
    proc seek(self: File, pos: I64) returns ref File
    /// Move the cursor by `delta` bytes relative to its current position
    /// (`delta` may be negative). Returns `self` so calls chain via UFCS.
    proc seek_cur(self: File, delta: I64) returns ref File
    /// Move the cursor to `delta` bytes relative to the end of the file
    /// (`delta` is typically <= 0). `f.seek_end(0)` followed by `f.tell()`
    /// yields the file size. Returns `self` so calls chain via UFCS.
    proc seek_end(self: File, delta: I64) returns ref File
    /// Close the underlying file handle. Private: invoked only from
    /// `delete`, which the garbager inserts after a File local's last
    /// use, so nothing reads the handle afterwards.
    func close(self: File)
    /// Release the File: close the handle, then free the struct shell
    /// unless the caller still owns it (`call_free = false` for a
    /// stack/value local). The garbager inserts this call right after the
    /// File's last use.
    func delete(own self: File, shallow call_free: Bool)
    proc fwrite(self: File, parts: ..Str) returns ref File
    func clone(self: File) returns File
    func size() returns shallow USize
}
#+end_src

* Dir
:PROPERTIES:
:CUSTOM_ID: til.src.std.file.Dir
:group: Standard library

A directory being read, entry by entry.

#+begin_src til
struct Dir {
    ref mut handle : U8
namespace:
    /// Start reading the directory at `path`. Exits with an error if it
    /// cannot be opened.
    proc open(path: Str) returns Dir
    /// The next entry's name (not its path), "" once every entry has
    /// been reported. "." and ".." are never returned. The order is
    /// whatever the file system gives -- sort if you need a stable one.
    proc next(self: Dir) returns Str
    /// Every entry in the directory at `path`, in file-system order.
    /// The whole listing at once, for callers that want to sort or
    /// filter it rather than stream it.
    proc list(path: Str) returns Vec__Str
    /// Close the underlying handle. Private: invoked only from `delete`,
    /// which the garbager inserts after a Dir local's last use.
    func close(self: Dir)
    /// Release the Dir: close the handle, then free the struct shell
    /// unless the caller still owns it.
    func delete(own self: Dir, shallow call_free: Bool)
    func clone(self: Dir) returns Dir
    func size() returns shallow USize
}
#+end_src
