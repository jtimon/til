:PROPERTIES:

File handle for reading or writing on disk.

#+begin_src til
struct File {
    ref mut handle : U8
    mut is_write : Bool
    mut is_update : Bool
namespace:
    proc readfile(path: Str) returns Str
    proc writefile(path: Str, content: Str)
    proc appendfile(path: Str, content: Str)
    proc remove(path: Str)
    proc exists(path: Str) returns shallow Bool
    proc is_dir(path: Str) returns shallow Bool
    proc is_file(path: Str) returns shallow Bool
    proc remove_tree(path: Str)
    proc rename(src: Str, dst: Str)
    proc new(path: Str, shallow is_write: Bool) returns File
    proc update(path: Str) returns File
    proc append(path: Str) returns File
    proc write(self: File, s: Str) returns ref File
    proc read_all(self: File) returns Str
    proc read(self: File, shallow count: I64) returns Str
    proc tell(self: File) returns shallow I64
    proc seek(self: File, pos: I64) returns ref File
    proc seek_cur(self: File, delta: I64) returns ref File
    proc seek_end(self: File, delta: I64) returns ref File
    func close(self: File)
    func delete(own self: File, shallow call_free: Bool)
    proc fwrite(self: File, parts: ..Str) returns ref File
    func clone(self: File) returns File
    func size() returns shallow USize
}
#+end_src

* File.remove
:PROPERTIES:
:CUSTOM_ID: til.src.std.file.File.remove
:group: Standard library
:END:

Delete the file at `path`, like `rm`. Removing a file the process
still has open is allowed on POSIX (the handle keeps working until
it is closed); a missing path is not an error. Directories are
not removed -- pointing this at one is an error.

#+begin_src til
proc File.remove(path: Str)
#+end_src

* File.remove_tree
:PROPERTIES:
:CUSTOM_ID: til.src.std.file.File.remove_tree
:group: Standard library
:END:

Remove `path` and everything under it, like `rm -rf`: a file, or a
directory with its whole subtree. A path that does not exist is
already removed and reports success; a symlink is unlinked, never
followed. Any other failure is fatal.

#+begin_src til
proc File.remove_tree(path: Str)
#+end_src

* File.rename
:PROPERTIES:
:CUSTOM_ID: til.src.std.file.File.rename
:group: Standard library
:END:

Move `src` onto `dst`, replacing whatever `dst` was. On POSIX the
replacement is atomic, which is what makes it safe to build a new
executable beside the running one and swap it in. Fatal on
failure -- notably when the two paths are on different file
systems, where a rename cannot happen and a copy is the answer.

#+begin_src til
proc File.rename(src: Str, dst: Str)
#+end_src

* File.append
:PROPERTIES:
:CUSTOM_ID: til.src.std.file.File.append
:group: Standard library
:END:

Open a file at `path` for appending: writes always land at the end
and existing contents are kept, where `File.new(path, true)`
truncates. The file is created when it does not exist. Reads are
rejected, and `seek` does not move where a write lands -- "ab"
appends regardless of the cursor.

#+begin_src til
proc File.append(path: Str) returns File
#+end_src
