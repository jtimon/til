* ReplEditor
:PROPERTIES:
:CUSTOM_ID: til.src.self.repl_editor.ReplEditor
:END:

Everything one interactive REPL owns: the line being edited, the
history it recalls from, and the reverse-search sub-mode. Constructed
once in run_repl and passed by mutation through every session, so a
~mode X~ restart keeps both the file's entries and this session's.

#+begin_src til
struct ReplEditor {
    mut interactive : Bool
    mut hist_path : Str
    mut persist : Bool
    own mut history : Vec__Str
    mut prompt : Str
    mut line : Str
    mut cursor : USize
    mut view_start : USize
    mut hist_idx : USize
    mut draft : Str
    mut searching : Bool
    mut search_query : Str
    mut search_idx : I64
    mut saved_line : Str
    mut saved_cursor : USize
namespace:
    func size() returns shallow USize
    func clone(self: ReplEditor) returns ReplEditor
    func delete(own self: ReplEditor, shallow call_free: Bool)
}
