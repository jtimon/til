enum Option__ref_HeapBinding {
    None
    Some: HeapBinding
namespace:
    func is_some(shallow self: Option__ref_HeapBinding) returns shallow Bool
    func is_none(shallow self: Option__ref_HeapBinding) returns shallow Bool
    func unwrap_or(self: Option__ref_HeapBinding, default_v: HeapBinding) returns ref HeapBinding
    /// The contained value; PANICS on None, ala Rust's unwrap. For the
    /// `if o.is_none() { ... }` + `o.unwrap()` idiom, where the caller
    /// has already handled absence and unwrapping cannot fail. Prefer
    /// a `switch` when both arms do real work, and `unwrap_or` when a
    /// default is meaningful -- reach for this only when None is
    /// unreachable by construction, so it fails loudly if that ever
    /// stops being true.
    func unwrap(self: Option__ref_HeapBinding) returns ref HeapBinding
    /// The contained value, MOVED out; PANICS on None, like unwrap.
    /// unwrap only BORROWS self, so for an owned payload it must hand
    /// back a deep clone; this consumes the Option instead, so the
    /// payload's bytes move and only the enum shell is freed (#321).
    /// Use it wherever the caller takes ownership of what the Option
    /// held -- an `Option(Expr)` returned by an AST helper, say, where
    /// a clone would copy a whole subtree and break node identity.
    func take(own self: Option__ref_HeapBinding) returns ref HeapBinding
    ext_func None() returns shallow Option__ref_HeapBinding
    ext_func Some(val: HeapBinding) returns shallow Option__ref_HeapBinding
    core_func is(self: Option__ref_HeapBinding, other: Option__ref_HeapBinding) returns shallow Bool
    func delete(own self: Option__ref_HeapBinding, shallow call_free: Bool)
    func to_str(shallow self: Option__ref_HeapBinding) returns Str
    func clone(self: Option__ref_HeapBinding) returns shallow Option__ref_HeapBinding
    func size() returns shallow USize

#+begin_src til
struct Map__Str_HeapBinding {
    mut keys : Vec__Str
    mut values : Vec__HeapBinding
namespace:
    /// Create an empty Map of the types provided for the key and values respectively.
    /// The key type must implement `cmp`.
    func new() returns Map__Str_HeapBinding
    /// Create an empty Map pre-sized for `n` entries, so inserting up to
    /// `n` reallocs neither backing Vec (Vec.with_capacity's
    /// counterpart). n=0 behaves like new(): lazy-empty, no allocation.
    func with_capacity(shallow n: USize) returns Map__Str_HeapBinding
    func len(self: Map__Str_HeapBinding) returns shallow USize
    /// Drop all live keys and values but keep the allocated storage so the Map can be reused.
    func clear(mut self: Map__Str_HeapBinding)
    /// True if a value exists for the given key.
    func has(self: Map__Str_HeapBinding, key: Str) returns shallow Bool
    /// Return a ref to the value for `key`. Throws KeyNotFound if
    /// `key` is absent; call as `map.get(k)!` (or `map[k]`) to panic
    /// on a missing key.
    func get(self: Map__Str_HeapBinding, key: Str, mut _err_kind: I64) returns ref HeapBinding
    /// Insert or overwrite the value for `key`. The value is owned.
    func set(mut self: Map__Str_HeapBinding, key: Str, own val: HeapBinding)
    /// Remove `key`, dropping its key and value. Returns false if absent.
    func remove(mut self: Map__Str_HeapBinding, key: Str) returns shallow Bool
    /// Entries in key order, in the `Name(k=v, k2=v2)` shape a struct's
    /// generated to_str uses. `keys`/`values` are Vecs, whose `own data`
    /// disqualifies Map from that generator, so this is hand-written.
